#!/bin/bash

PROJECT_PATH="$1"
NODE_VER="$2"
APP_JS="$3"

export NVM_DIR="/usr/nvm"
[ -s "$NVM_DIR/nvm.sh" ] && \. "$NVM_DIR/nvm.sh"  # This loads nvm

if ! nvm use "$NODE_VER"; then
        echo "Node version $NODE_VER is not available. Exiting." 1>&2
        exit 1
fi

cd "$PROJECT_PATH"

if ! npm ls unit-http; then
        npm link unit-http 2>/dev/null || :
fi

NODE_NO_WARNINGS=1 node --loader unit-http/loader.mjs --require unit-http/loader "$APP_JS"
